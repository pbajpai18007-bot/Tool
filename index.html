async function searchAudio() { const query = document.getElementById('songQuery').value.trim(); const btn = document.getElementById('searchBtn'); const songTitle = document.getElementById('songTitle'); const songArtist = document.getElementById('songArtist'); const poster = document.getElementById('poster'); const audio = document.getElementById('audioPlayer'); if (!query) { alert('Pehle gaane ka naam likho!'); return; } btn.innerText = 'Searching... ⏳'; songTitle.innerText = "Finding full song..."; songArtist.innerText = query; let foundSong = false; // ENGINE 1: YouTube Audio Proxy (Guaranteed Full Song & No 30s limit) const pipedApis = [ 'https://pipedapi.kavin.rocks', 'https://pipedapi.smnz.de', 'https://pipedapi.tokhmi.xyz' ]; for (const api of pipedApis) { if (foundSong) break; try { // 1. YouTube par gaana search karega const searchRes = await fetch(`${api}/search?q=${encodeURIComponent(query + ' official audio')}&filter=all`); const searchData = await searchRes.json(); if (searchData.items && searchData.items.length > 0) { // Pehla streamable video track nikalna const track = searchData.items.find(item => item.type === 'stream'); if (!track) continue; // Video ID extract karna (e.g., /watch?v=xyz123) const videoId = track.url.split('v=')[1] || track.url.replace('/watch?v=', ''); songTitle.innerText = track.title; songArtist.innerText = track.uploaderName || "YouTube Audio"; poster.src = track.thumbnail; // 2. Us video ka direct high-quality Audio Stream URL nikalna const streamRes = await fetch(`${api}/streams/${videoId}`); const streamData = await streamRes.json(); if (streamData.audioStreams && streamData.audioStreams.length > 0) { // Sabse best audio bitrate select karna const bestAudio = streamData.audioStreams.sort((a, b) => b.bitrate - a.bitrate)[0]; audio.src = bestAudio.url; audio.load(); const playPromise = audio.play(); if (playPromise !== undefined) { playPromise.catch(() => { songArtist.innerText = "Click Play Button Below ▶"; }); } foundSong = true; } } } catch (e) { console.log("YouTube API failed, trying next...", e); } } // ENGINE 2: JioSaavn Latest Backup (Agar YouTube proxy slow ho) if (!foundSong) { try { const saavnRes = await fetch(`https://saavn.dev/api/search/songs?query=${encodeURIComponent(query)}`); const saavnData = await saavnRes.json(); const list = saavnData?.data?.results || []; if (list.length > 0) { const track = list[0]; songTitle.innerText = track.name.replace(/"/g, '"').replace(/'/g, "'").replace(/&/g, "&"); songArtist.innerText = track.primaryArtists || track.singers || "Top Track"; poster.src = track.image[track.image.length - 1]?.url || track.image[0]?.url; let streamUrl = track.downloadUrl[track.downloadUrl.length - 1]?.url; if (streamUrl) { audio.src = streamUrl; audio.load(); const playPromise = audio.play(); if (playPromise !== undefined) { playPromise.catch(() => { songArtist.innerText = "Click Play Button Below ▶"; }); } foundSong = true; } } } catch (e) { console.log("Saavn Fallback Failed: ", e); } } if (!foundSong) { alert("गाना नहीं मिल पाया। कृपया स्पेलिंग चेक करें या कोई दूसरा गाना ट्राई करें!"); songTitle.innerText = "Song not found"; songArtist.innerText = "Try another search"; } btn.innerText = 'Play Song'; }
